const SPREADSHEET_ID = '1iIWQ_Kbqk3vraXWxan2SwBiRGhGD5YI9ODfTTJJmuVY'; function doGet(e) { if (e && e.parameter && e.parameter.mode === 'json') { const data = getAppData(); return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } return HtmlService.createHtmlOutputFromFile('Index') .setTitle('Tes Formatif PJOK') .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const result = submitAnswers(payload); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function getAppData() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); // 1. Ambil Pengaturan const sheetPengaturan = ss.getSheetByName('Pengaturan'); if (!sheetPengaturan) throw new Error("Sheet 'Pengaturan' tidak ditemukan."); const dataPengaturan = sheetPengaturan.getRange(2, 1, 3, 2).getValues(); const pengaturan = { materiUmum: dataPengaturan[0][0] || 'Tes Formatif PJOK', backgroundUrl: dataPengaturan[0][1] || 'https://images.unsplash.com/photo-1451187580459-43490279c0fa?q=80&w=2072&auto=format&fit=crop', token: (dataPengaturan[1] && dataPengaturan[1][1]) ? dataPengaturan[1][1].toString().trim() : '12345', waktu: (dataPengaturan[2] && dataPengaturan[2][1]) ? parseInt(dataPengaturan[2][1], 10) : 60 }; // 2. Ambil Data Siswa const sheetSiswa = ss.getSheetByName('DataSiswa'); if (!sheetSiswa) throw new Error("Sheet 'DataSiswa' tidak ditemukan."); const dataSiswaRaw = sheetSiswa.getDataRange().getValues(); dataSiswaRaw.shift(); const kelasSet = new Set(); const siswaData = []; dataSiswaRaw.forEach(row => { if (row[0] && row[1]) { const kelas = row[0].toString().trim(); const nama = row[1].toString().trim(); const absen = row[2] ? row[2].toString().trim() : '-'; kelasSet.add(kelas); siswaData.push({ kelas, nama, absen }); } }); // 3. Ambil Soal & Daftar Materi (selalu baca isi terbaru sheet 'Soal') const sheetSoal = ss.getSheetByName('Soal'); if (!sheetSoal) throw new Error("Sheet 'Soal' tidak ditemukan."); const dataSoalRaw = sheetSoal.getDataRange().getValues(); dataSoalRaw.shift(); const soal = []; const materiSet = new Set(); dataSoalRaw.forEach((row, index) => { if (row[3]) { const materiTarget = row[2] ? row[2].toString().trim() : 'Umum'; if (materiTarget !== '') { materiSet.add(materiTarget); } soal.push({ idSoal: index, no: row[0], kelasTarget: row[1] ? row[1].toString().trim() : 'Semua', materiTarget: materiTarget, pertanyaan: row[3].toString(), opsi: { A: row[4] ? row[4].toString() : '', B: row[5] ? row[5].toString() : '', C: row[6] ? row[6].toString() : '', D: row[7] ? row[7].toString() : '', E: row[8] ? row[8].toString() : '' }, gambar: row[10] ? row[10].toString().trim() : '' }); } }); return { pengaturan: pengaturan, kelas: Array.from(kelasSet).sort(), materiList: Array.from(materiSet).sort(), siswaData: siswaData, soal: soal }; } function submitAnswers(payload) { try { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheetJawaban = ss.getSheetByName('Jawaban'); if (!sheetJawaban) throw new Error("Sheet 'Jawaban' tidak ditemukan."); const sheetSoal = ss.getSheetByName('Soal'); if (!sheetSoal) throw new Error("Sheet 'Soal' tidak ditemukan."); const dataSoal = sheetSoal.getDataRange().getValues(); dataSoal.shift(); let benar = 0; const totalSoal = payload.jawaban.length; payload.jawaban.forEach((item) => { const rowData = dataSoal[item.idSoal]; const kunci = rowData && rowData[9] ? rowData[9].toString().toUpperCase().trim() : ''; if (item.jawaban && kunci && item.jawaban.toUpperCase() === kunci) { benar++; } }); const skor = totalSoal > 0 ? ((benar / totalSoal) * 100).toFixed(2) : '0.00'; const timestamp = new Date(); const newRow = [ timestamp, payload.status, skor, payload.identitas.materi, payload.identitas.kelas, payload.identitas.nama, payload.identitas.absen ]; payload.jawaban.forEach(j => { newRow.push(j.jawaban); }); sheetJawaban.appendRow(newRow); return { success: true, message: 'Jawaban tersimpan. Nilai: ' + skor }; } catch (error) { return { success: false, message: error.toString() }; } }
